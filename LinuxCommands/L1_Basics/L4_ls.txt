	Blue color 	= directories files
	Green color = executable files

ls = Long List
 
	See the list with file name ascending order  ?

		ls -l

	See the list file name on descending order ?

		ls -lr
	
	See the list based on time descending order the  newest file ?

		ls -lt 

	See the list based on time descending order the oldest file ?

		ls -ltr

	See the Memory with MB or human readable ?

		ls -ltrh
	
	See the multiple folder of file or recursive file ?

		ls -R

	See the hidden files ?
		
		ls -a
	
	See the Recursive and hidden file ?

		ls -Ra
