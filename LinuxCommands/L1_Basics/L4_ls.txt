#Note :

	Blue color 	= directories files
	Green color = executable files 
	Black Color = normal files


#Summary of ls
	l = filename
	t = new file
	S = Space
	h = human readable
	a = hidden file


ls = Long List
 
	file name ascending order  ?
		ls -l

	file name on descending order ?
		ls -lr
		#Note r is reverse
	
	newest file ?
		ls -lt 
		#Note Where t represents the newes(t)

	oldest file ?
		ls -ltr

	size of file ?
		ls -lS
		#Note : Not this one ls -ls
	
	size in reverse ?
		ls -lSr

	See the Memory with MB or human readable ?
		ls -lth
		ls -ltrh
	
	See the multiple folder of file or recursive file ?
		ls -R

	See the hidden files ?
		ls -a
