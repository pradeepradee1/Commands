#Summary of ls
	l = filename
	t = new file
	S = Space
	h = human readable
	a = hidden file
	r = reverse
	R = Recurrsive


List Files ?		
		hdfs dfs -ls /
		hdfs dfs -ls pradeep/

filename ?
		hdfs dfs -ls -l

newest file ?		
		hdfs dfs -ls -t

Sort the Files based on the  size ?
		hdfs dfs -ls -S
		hdfs dfs -ls -S -h (human readable)

Sort the Files in reverse order ?
		hdfs dfs -ls -r

see the recursive file ?
		hdfs dfs -ls -R

See the files size ?
		hdfs dfs -ls -S
		hdfs dfs -ls -S -h (human readable)

List Files Directory Name ?
		hdfs dfs -ls -C
